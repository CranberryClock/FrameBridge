cmake_minimum_required(VERSION 3.31)
project(framebridge-null-runtime LANGUAGES CXX)

add_library(framebridge-temporal STATIC native/cube-renderer/temporal_frame.cpp)
target_include_directories(framebridge-temporal PUBLIC native/cube-renderer native/mirror-receiver native/bridge)
target_compile_features(framebridge-temporal PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-temporal PRIVATE /W4 /WX /permissive-)
endif()
target_link_libraries(framebridge-temporal PUBLIC d3d12 d3dcompiler dxgi)
add_executable(framebridge-temporal-test native/cube-renderer/temporal_test.cpp)
target_link_libraries(framebridge-temporal-test PRIVATE framebridge-temporal)
target_compile_features(framebridge-temporal-test PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-temporal-test PRIVATE /W4 /WX /permissive-)
endif()
add_test(NAME framebridge-temporal-contract COMMAND framebridge-temporal-test)
add_test(NAME framebridge-temporal-negative-control COMMAND framebridge-temporal-test --fail)
set_tests_properties(framebridge-temporal-negative-control PROPERTIES WILL_FAIL TRUE)

set(IXWEBSOCKET_INSTALL OFF CACHE BOOL "" FORCE)
set(USE_TLS OFF CACHE BOOL "" FORCE)
set(USE_ZLIB OFF CACHE BOOL "" FORCE)
add_subdirectory(third_party/ixwebsocket EXCLUDE_FROM_ALL)
include(third_party/ixwebsocket-limits.cmake)
add_executable(framebridge-null-runtime native/runtime/src/main.cpp)
target_compile_features(framebridge-null-runtime PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-null-runtime PRIVATE /W4 /permissive-)
endif()

add_executable(framebridge-bridge-codec native/bridge/codec_test.cpp)
target_include_directories(framebridge-bridge-codec PRIVATE native/bridge)
target_compile_features(framebridge-bridge-codec PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-bridge-codec PRIVATE /W4 /WX /permissive-)
endif()
enable_testing()
add_test(NAME framebridge-codec COMMAND framebridge-bridge-codec)
set_tests_properties(framebridge-codec PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")

  native/streamline-preflight/preflight.cpp)

add_library(framebridge-native-mirror-session STATIC
  native/mirror-receiver/mirror_session.cpp)
target_include_directories(framebridge-native-mirror-session PUBLIC
  native/mirror-receiver native/bridge third_party/nlohmann-json/single_include)
target_compile_features(framebridge-native-mirror-session PUBLIC cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-native-mirror-session PRIVATE /W4 /WX /permissive-)
endif()

add_executable(framebridge-native-mirror native/mirror-receiver/main.cpp)
target_link_libraries(framebridge-native-mirror PRIVATE framebridge-native-mirror-session ixwebsocket ws2_32)
if(DAWN_ROOT)
  include(native/cube-renderer/dawn-link.cmake)
  target_link_libraries(framebridge-native-mirror PRIVATE framebridge-renderer)
  target_compile_definitions(framebridge-native-mirror PRIVATE FRAMEBRIDGE_HAS_DAWN=1)
  add_executable(framebridge-native-cube native/cube-renderer/main.cpp)
  target_link_libraries(framebridge-native-cube PRIVATE framebridge-renderer)
endif()

target_compile_definitions(framebridge-native-mirror PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
if(MSVC)
  target_compile_options(framebridge-native-mirror PRIVATE /W4 /WX /permissive-)
endif()

add_executable(framebridge-native-mirror-test native/mirror-receiver/mirror_session_test.cpp)
