cmake_minimum_required(VERSION 3.31)
project(framebridge-null-runtime LANGUAGES CXX)
add_executable(framebridge-null-runtime native/runtime/src/main.cpp)
target_compile_features(framebridge-null-runtime PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-null-runtime PRIVATE /W4 /permissive-)
endif()

add_executable(framebridge-bridge-codec native/bridge/codec_test.cpp)
target_include_directories(framebridge-bridge-codec PRIVATE native/bridge)
target_compile_features(framebridge-bridge-codec PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(framebridge-bridge-codec PRIVATE /W4 /WX /permissive-)
endif()
enable_testing()
add_test(NAME framebridge-codec COMMAND framebridge-bridge-codec)
set_tests_properties(framebridge-codec PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
